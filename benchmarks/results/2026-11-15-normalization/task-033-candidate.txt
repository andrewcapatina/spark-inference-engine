import torch
import triton
import triton.language as tl

@triton.jit
def _norm_tail_kernel(x_ptr, inv_ptr, w_ptr, out_ptr, n_elements, WIDTH: tl.constexpr, BLOCK: tl.constexpr):
    offsets = tl.program_id(0) * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < n_elements
    rows = offsets // WIDTH
    cols = offsets % WIDTH
    x = tl.load(x_ptr + offsets, mask=mask, other=0.0).to(tl.float32)
    inv = tl.load(inv_ptr + rows, mask=mask, other=0.0).to(tl.float32)
    w = tl.load(w_ptr + cols, mask=mask, other=0.0).to(tl.float32)
    out = (x * inv) * (w + 1.0)
    tl.store(out_ptr + offsets, out.to(tl.bfloat16), mask=mask)

def norm_tail(x_fp32, inverse_rms, weight):
    assert torch.is_inference_mode_enabled()
    assert x_fp32.is_cuda and inverse_rms.is_cuda and weight.is_cuda
    assert x_fp32.is_contiguous() and inverse_rms.is_contiguous() and weight.is_contiguous()
    assert x_fp32.dtype == torch.float32 and inverse_rms.dtype == torch.float32
    assert weight.dim() == 1 and weight.dtype in (torch.bfloat16, torch.float32)
    assert x_fp32.dim() >= 1 and x_fp32.shape[-1] > 0
    width = x_fp32.shape[-1]
    assert weight.shape[0] == width
    assert inverse_rms.numel() == x_fp32.numel() // width
    out = torch.empty_like(x_fp32, dtype=torch.bfloat16)
    if x_fp32.numel() == 0:
        return out
    grid = (triton.cdiv(x_fp32.numel(), 1024),)
    _norm_tail_kernel[grid](x_fp32, inverse_rms, weight, out, x_fp32.numel(),
                            WIDTH=width, BLOCK=1024, num_warps=4, enable_fp_fusion=False)
    return out
