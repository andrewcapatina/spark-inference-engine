def gated_rms_norm(x, weight, z, eps=1e-6):
    """Qwen3.5 gated RMSNorm: norm(x) * weight * silu(z)."""
    dtype = x.dtype
    x = x.float()
    variance = x.pow(2).mean(-1, keepdim=True)
    x = x * torch.rsqrt(variance + eps)
    x = weight * x.to(dtype)
    z = torch.nn.functional.silu(z.to(torch.float32))
    return (x * z).to(dtype)
