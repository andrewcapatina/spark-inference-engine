import torch

def solve_pair(system, v_beta, decayed_k):
    rhs = torch.cat([v_beta, decayed_k], dim=-1)
    sol = torch.linalg.solve_triangular(system, rhs, upper=False, unitriangular=True)
    v_new = sol[..., :v_beta.shape[-1]]
    k_cumdecay = sol[..., v_beta.shape[-1]:]
    return v_new, k_cumdecay
