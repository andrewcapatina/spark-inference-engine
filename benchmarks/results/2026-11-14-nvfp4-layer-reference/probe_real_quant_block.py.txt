"""Diagnostic-only single-block fixture and independent real-projection checks."""
from pathlib import Path
import json,gc,weakref,importlib.util
import torch
from safetensors import safe_open
from layer_reference_quant import activation_qdq,unpack_nvfp4
from compressed_tensors.quantization import QuantizationArgs
from compressed_tensors.quantization.lifecycle.forward import quantize
from compressed_tensors.quantization.utils.helpers import compute_dynamic_scales_and_zp

def run(out,save,fixture):
    from spark_engine.qwen_model import QwenDecoderLayer
    from spark_engine.qwen_fp8 import FP8Linear
    from spark_engine.qwen_nvfp4 import NVFP4Linear
    import flashinfer
    cfgfull=json.loads(Path('/model/config.json').read_text());cfg=cfgfull['text_config'];i=int(fixture.get('probe_layer',1));prefix=f'model.language_model.layers.{i}.'
    index=json.loads(Path('/model/model.safetensors.index.json').read_text())['weight_map'];handles={s:safe_open('/model/'+s,framework='pt',device='cpu') for s in set(index.values())}
    def get(name):return handles[index[prefix+name]].get_tensor(prefix+name)
    with torch.device('meta'):block=QwenDecoderLayer(cfg,cfg['layer_types'][i])
    for name,module in list(block.named_modules()):
        if not isinstance(module,torch.nn.Linear):continue
        if prefix+name+'.weight_packed' in index:replacement=NVFP4Linear(*[get(name+'.'+suffix).cuda() for suffix in ('weight_packed','weight_scale','weight_global_scale','input_global_scale')])
        elif get(name+'.weight').dtype==torch.float8_e4m3fn:replacement=FP8Linear(get(name+'.weight').cuda(),get(name+'.weight_scale').cuda())
        else:continue
        parent,_,leaf=name.rpartition('.');setattr(block.get_submodule(parent) if parent else block,leaf,replacement)
    for name,param in list(block.named_parameters()):
        value=get(name).cuda().to(torch.bfloat16);assert value.shape==param.shape;parent,_,leaf=name.rpartition('.');setattr(block.get_submodule(parent) if parent else block,leaf,torch.nn.Parameter(value,requires_grad=False))
    block.eval();qa8=QuantizationArgs.model_validate(cfgfull['quantization_config']['config_groups']['group_0']['input_activations']);qa4=QuantizationArgs.model_validate(cfgfull['quantization_config']['config_groups']['group_1']['input_activations'])
    d={'status':'running','mode':fixture.get('probe_mode','probe'),'layer':i,'scope':'Actual captured-block projection inputs: independent CT-QDQ/FP32 matmul with fixed existing .01/.03125 bounds; source engine unchanged','checks':[]};save(d)
    def error(a,b):
        a,b=a.float(),b.float();delta=(a-b).abs();bound=.03125+.01*b.abs()
        return {'max_abs':delta.max().item(),'rms':delta.square().mean().sqrt().item(),'relative_l2':(delta.norm()/b.norm().clamp_min(1e-20)).item(),'outside_existing_bound':(delta>bound).sum().item(),'elements':a.numel()}
    hooks=[]
    def make_hook(name):
        def hook(module,args,actual):
            x=args[0];x2d=x.reshape(-1,x.shape[-1]);row={'name':name,'shape':list(x.shape),'kind':type(module).__name__}
            if isinstance(module,NVFP4Linear):
                weight=unpack_nvfp4(module.weight_packed,module.weight_scale,module.weight_global_scale);qdq=activation_qdq(x,qa4,module.input_global_scale).reshape_as(x2d)
                packed,scale=flashinfer.fp4_quantize(x2d,module.input_global_scale,is_sf_swizzled_layout=False);scale=scale.view(torch.float8_e4m3fn).reshape(x2d.shape[0],x2d.shape[1]//16);kernel_qdq=unpack_nvfp4(packed,scale,module.input_global_scale)
                row['activation_qdq_error']=error(kernel_qdq,qdq);different=(kernel_qdq!=qdq).nonzero();row['different_FP32_reconstructions']=different.shape[0];row['input_global_scale']=module.input_global_scale.item()
                ref_scale,ref_zp=compute_dynamic_scales_and_zp(x2d.float().unsqueeze(0),qa4,None,global_scale=module.input_global_scale)
                ref_levels=quantize(x2d.float().unsqueeze(0),ref_scale,ref_zp,qa4,dtype=torch.float32,global_scale=module.input_global_scale).reshape_as(x2d)
                kernel_levels=unpack_nvfp4(packed,torch.ones_like(scale.float()).to(torch.float8_e4m3fn),torch.ones(1,device='cuda'))
                row['fp4_level_disagreements']=(ref_levels!=kernel_levels).sum().item();row['local_scale_disagreements']=(ref_scale.float().reshape_as(scale)!=scale.float()).sum().item()
                different=(ref_levels!=kernel_levels).nonzero()
                row['activation_examples']=[{'row':a,'col':b,'input':x2d[a,b].item(),'independent_qdq':qdq[a,b].item(),'kernel_qdq':kernel_qdq[a,b].item()} for a,b in different[:8].tolist()]
            else:
                weight=module.weight.float()*module.weight_scale.float();qdq=activation_qdq(x,qa8).reshape_as(x2d);scale=(x2d.float().abs().amax(-1,keepdim=True)/448).clamp_min(1e-6);aq=(x2d/scale).to(torch.float8_e4m3fn).contiguous();kernel_qdq=aq.float()*scale
                row['activation_qdq_error']=error(kernel_qdq,qdq)
                # Diagnostic direct call only; never placed in engine implementation.
                pad=(-x2d.shape[0])%16
                aq=torch.nn.functional.pad(aq,(0,0,0,pad));sa=torch.nn.functional.pad(scale,(0,0,0,pad))
                slow=torch._scaled_mm(aq,module.weight.T,sa,module.weight_scale.T,out_dtype=torch.bfloat16,use_fast_accum=False)[:x2d.shape[0]].reshape_as(actual)
                row['fast_vs_regular_accum']=error(actual,slow);del slow,aq,sa
            expected=(qdq@weight.T).to(torch.bfloat16).reshape_as(actual);same_quant=(kernel_qdq@weight.T).to(torch.bfloat16).reshape_as(actual)
            row['native_vs_independent']=error(actual,expected);row['gemm_with_same_quantized_input']=error(actual,same_quant);d['checks'].append(row);save(d);print(json.dumps(row),flush=True)
        return hook
    for name,module in block.named_modules():
        if isinstance(module,(NVFP4Linear,FP8Linear)):hooks.append(module.register_forward_hook(make_hook(name)))
    captured=torch.load(Path('/results')/fixture['capture_run']/f'fresh-layer-{i:03d}.pt',map_location='cuda',weights_only=True)
    with torch.inference_mode():
        if cfg['layer_types'][i]=='full_attention':
            from transformers.models.qwen3_5.configuration_qwen3_5 import Qwen3_5TextConfig
            from transformers.models.qwen3_5.modeling_qwen3_5 import Qwen3_5TextRotaryEmbedding
            rotary=Qwen3_5TextRotaryEmbedding(Qwen3_5TextConfig(**cfg),device='cuda');n=captured['input'].shape[1];pos=torch.arange(n,device='cuda')[None,None,:].expand(3,1,-1);rope=rotary(captured['input'],pos);mask=torch.full((n,n),float('-inf'),device='cuda',dtype=torch.bfloat16).triu(1);result,cache=block.prefill(captured['input'],rope,mask);del rotary,pos,rope,mask
        else:result,cache=block.prefill(captured['input'])
    d['single_block_reproduces_captured_last']=torch.equal(result[:,-1:],captured['output_last']);d['block_reproduction_error']=error(result[:,-1:],captured['output_last']);assert d['single_block_reproduces_captured_last']
    for hook in hooks:hook.remove()
    del hook,hooks,block,module,replacement,param,value,captured,result,cache;handles.clear();gc.collect();torch.cuda.synchronize();torch._C._cuda_clearCublasWorkspaces()
    import flashinfer.utils
    flashinfer.utils._cache_buf.clear()
    retired=getattr(flashinfer.utils,'_cache_buf_retired',None)
    if isinstance(retired,list):retired.clear()
    gc.collect();torch.cuda.empty_cache();torch.cuda.synchronize();d['released_memory']={'allocated_bytes':torch.cuda.memory_allocated(),'reserved_bytes':torch.cuda.memory_reserved()};save(d);assert d['released_memory']['allocated_bytes']<=16*1024**2 and d['released_memory']['reserved_bytes']<=64*1024**2
    d['status']='complete';save(d);print('REAL_BLOCK_PROJECTION_DIAGNOSTIC_COMPLETE',flush=True)
