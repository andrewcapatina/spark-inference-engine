"""Streaming reader for Qwen compressed-tensors mixed-precision checkpoints.

Native cache remains BF16; this module only streams tensors from disk.
"""
import json
from pathlib import Path
from contextlib import contextmanager, ExitStack
from safetensors import safe_open
import torch

@contextmanager
def quantized_checkpoint(directory):
    root = Path(directory).resolve()
    cfg = json.loads((root / "config.json").read_text())
    qc = cfg.get("quantization_config", {})
    if qc.get("quant_method") != "compressed-tensors" or qc.get("format") != "mixed-precision":
        raise ValueError("not a compressed-tensors mixed-precision checkpoint")
    text_cfg = cfg.get("text_config", cfg)
    index = json.loads((root / "model.safetensors.index.json").read_text())
    weight_map = index["weight_map"]
    canonical = {}
    for name, shard in weight_map.items():
        if name.startswith("model.visual.") or name.startswith("mtp.") or name.startswith("model.mtp."):
            continue
        if name.startswith("model.language_model."):
            canon = name[len("model.language_model."):]
        elif name.startswith("lm_head."):
            canon = name
        else:
            raise ValueError(f"unexpected tensor namespace: {name}")
        if canon.endswith(".self_attn.k_scale") or canon.endswith(".self_attn.v_scale"):
            continue
        if canon in canonical:
            raise ValueError(f"duplicate canonical key: {canon}")
        canonical[canon] = (name, shard)
    shards = {}
    with ExitStack() as stack:
        for _, (_, shard) in canonical.items():
            if shard not in shards:
                path = root / shard
                if not path.resolve().is_relative_to(root):
                    raise ValueError(f"shard escapes directory: {shard}")
                shards[shard] = stack.enter_context(safe_open(str(path), framework="pt", device="cpu"))
        consumed = set()
        def get_tensor(canon):
            orig, shard = canonical[canon]
            t = shards[shard].get_tensor(orig)
            consumed.add(canon)
            return t
        yield text_cfg, get_tensor, set(canonical)
        missing = set(canonical) - consumed
        if missing:
            raise RuntimeError(f"unconsumed tensors: {sorted(missing)[:5]}")

def load_quantized_qwen(directory, dtype=torch.bfloat16, device='cuda'):
    if dtype != torch.bfloat16:
        raise ValueError("only bfloat16 dtype supported")
    if not device.startswith('cuda'):
        raise ValueError("only cuda device supported")
    from spark_engine.qwen import QwenCausalLM
    from spark_engine.nvfp4 import NVFP4Linear
    from spark_engine.fp8 import FP8Linear
    nvfp4_count = 0
    fp8_count = 0
    with quantized_checkpoint(directory) as (cfg, get, keys):
        model = QwenCausalLM(cfg)
        with torch.device('meta'):
            pass
        for name, module in list(model.named_modules()):
            if not isinstance(module, torch.nn.Linear):
                continue
            packed_key = name + '.weight_packed'
            if packed_key in keys:
                wp = get(packed_key).to(device=device)
                ws = get(name + '.weight_scale').to(device=device)
                wgs = get(name + '.weight_global_scale').to(device=device)
                igs = get(name + '.input_global_scale').to(device=device)
                if wp.shape != (module.out_features, module.in_features // 2):
                    raise ValueError(f"NVFP4 packed shape mismatch at {name}")
                repl = NVFP4Linear(wp, ws, wgs, igs)
                nvfp4_count += 1
            else:
                wkey = name + '.weight'
                if wkey not in keys:
                    continue
                w = get(wkey)
                if w.dtype == torch.float8_e4m3fn:
                    if w.shape != (module.out_features, module.in_features):
                        raise ValueError(f"FP8 shape mismatch at {name}")
                    s = get(name + '.weight_scale').to(device=device)
                    w = w.to(device=device)
                    repl = FP8Linear(w, s)
                    fp8_count += 1
                else:
                    continue
            parent_path = '.'.join(name.split('.')[:-1])
            child_name = name.split('.')[-1]
            if parent_path:
                parent = model.get_submodule(parent_path)
            else:
                parent = model
            setattr(parent, child_name, repl)
        for name, param in list(model.named_parameters()):
            if param.device.type == 'meta':
                canon = name
                if canon not in keys:
                    raise KeyError(f"missing tensor for {canon}")
                t = get(canon)
                if t.shape != param.shape:
                    raise ValueError(f"shape mismatch at {canon}: {t.shape} vs {param.shape}")
                if t.dtype not in (torch.bfloat16, torch.float16, torch.float32):
                    raise ValueError(f"unexpected dtype at {canon}: {t.dtype}")
                new_p = torch.nn.Parameter(t.to(device=device, dtype=dtype), requires_grad=False)
                parent_path = '.'.join(name.split('.')[:-1])
                child_name = name.split('.')[-1]
                if parent_path:
                    parent = model.get_submodule(parent_path)
                else:
                    parent = model
                setattr(parent, child_name, new_p)
        model.quantization_metadata = {
            'format': 'nvfp4_fp8',
            'kv_cache_dtype': 'bfloat16',
            'nvfp4_linears': nvfp4_count,
            'fp8_linears': fp8_count,
        }
    return model.eval()
